BASIC COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

dir (ls)
→ list files and folders

cd (cd)
→ show/change current directory

cls (clear)
→ clear terminal screen

cd folderName (cd folderName)
→ change directory

cd .. (cd ..)
→ one step back

cd ..\.. (cd ../..)
→ two steps back

mkdir folderName (mkdir folderName)
→ create folder

type nul > file.txt (touch file.txt)
→ create file

del file.txt (rm file.txt)
→ delete file

rmdir folderName (rmdir folderName)
→ remove empty folder

rmdir /s /q folderName (rm -rf folderName)
→ remove folder with all contents permanently

copy file1 file2 (cp file1 file2)
→ copy file

move oldName newName (mv oldName newName)
→ move or rename

type file.txt (cat file.txt)
→ show file contents

echo Hello (echo Hello)
→ print text

doskey /history (history)
→ show command history

exit (exit)
→ close terminal


EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

mkdir project

type nul > index.html
(touch index.html)

cd project

cd ..

del app.js
(rm app.js)

rmdir /s /q project
(rm -rf project)


IMPORTANT NOTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

del / rm permanently delete files.

rmdir /s /q
and
rm -rf

permanently delete folders and contents.

Files do NOT go to Recycle Bin / Trash.


PATHS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Absolute Path
→ starts from root/drive

Windows:
D:\Coding\Java

Linux:
/home/imran/projects

Absolute path works from anywhere.


2. Relative Path
→ starts from current directory

Windows:
Projects\Test

Linux:
projects/test

Relative path depends on current location.
