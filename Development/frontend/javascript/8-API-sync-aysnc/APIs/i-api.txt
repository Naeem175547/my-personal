async function getData() {

    // Cat Fact
    let catRes = await fetch("https://catfact.ninja/fact");
    let catData = await catRes.json();
    console.log("Cat Fact:", catData.fact);

    // Activity
    let actRes = await fetch("https://www.boredapi.com/api/activity");
    let actData = await actRes.json();
    console.log("Activity:", actData.activity);

    // Dog Image
    let dogRes = await fetch("https://dog.ceo/api/breeds/image/random");
    let dogData = await dogRes.json();
    console.log("Dog Image:", dogData.message);
}

🔥 QUERY STRING (SHORT NOTES) 🔥
═══════════════════════════════════════
✔ Query String = Extra data in URL
Syntax:
URL?key=value

Example:
https://api.com/users?id=1
───────────────────────────────────────
✔ Multiple Parameters
https://api.com/users?id=1&name=imran

?  → Starts query string
&  → Separates parameters

───────────────────────────────────────

✔ Common Uses

✅ Search
✅ Filter
✅ Pagination
✅ API requests

───────────────────────────────────────
✔ JavaScript Read Query String

let params = new URLSearchParams(window.location.search);

console.log(params.get("name"));

───────────────────────────────────────

✔ Important Methods

get()     → Get value
has()     → Check parameter
append()  → Add parameter
delete()  → Remove parameter

───────────────────────────────────────

✔ Example

let params = new URLSearchParams();
params.append("name", "imran");
console.log(params.toString());

Output:
name=imran

───────────────────────────────────────

✔ Query String vs Route Parameter

?id=1      → Query String
/users/1   → Route Parameter

───────────────────────────────────────

✔ encodeURIComponent()

Used for special characters.

encodeURIComponent("New York")

Output:
New%20York
